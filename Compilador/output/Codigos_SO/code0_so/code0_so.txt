void main(void){
    int programas[10];
    int posicoes[10];

    int comando;
    int numeroPrograma;
    int iPrograma;
    
    int proxPrograma;
    int proxPosicao;

    int estadoPrograma;

    int i;

    /* inicializando os valores */
    i = 0;
    while(i < 10){
        programas[i] = i + 1;
        posicoes[i] = i + 1;
        i = i + 1;
    }

    while(1 == 1){
        showLCD(0); /* Orientacoes */
        comando = input();

        if(comando == 0){
            showLCD(1);
            numeroPrograma = input();
            
            /* Busca pelo programa */
            i = 0;
            iPrograma = 0 - 1;
            while(i < 10){
                if(programas[i] == numeroPrograma){
                    iPrograma = i;
                } 
                i = i + 1;
            }

            definirNumeroDoPrograma(numeroPrograma);
            if(iPrograma != 0 - 1){
                definirQuantum(0); /* Quantum infinito */
                showLCD(4); /* Executando P[] */
                proxPrograma = numeroPrograma;
                proxPosicao = posicoes[iPrograma];

                /* Troca de contexto */
                definirPosicaoDoPrograma(proxPosicao);
                saveRegs();
                atualizarMADMP();
                loadRegs();
                trocarDeContexto();

                /* Retorno do programa */
                saveRegs();
                atualizarMADMP();
                loadRegs();
                
                estadoPrograma = lerReg(11);
                if(estadoPrograma == 0){
                    showLCD(11); /* P[] chegou ao fim */
                    output(0);
                }
                else{
                    showLCD(12); /* Erro fatal - Nao deve ocorrer */
                    output(0);
                }
            }
            /* Caso nao encontre */
            else{
                showLCD(8); /* P[] nao encontrado */
                output(0);
            }
        }
        else{
            comando = input();
        }
    }
}
