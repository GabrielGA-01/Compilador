int c;
int d[5];
int e;
int f[5];

void alinha(int vet[]){
    vet[0] = c;
    vet[1] = d[2];
    d[3] = 3;
    
    e = f[2];
}

int seguro(){
    return(f[3]+c);
}

void main(void)
{
    int vetor[4];
    int gh;
    int bla[1];
    alinha(vetor);

    while(vetor[0] < 2){
        while(vetor[1] > 3){
            if(vetor[2] == 2){
                vetor[1] = vetor[1] - 1;
                e = vetor[1];
            }
            else{
                vetor[1] = vetor[1] - 2;
            }
            gh = seguro();
        }
        vetor[0] = 4 - e;
    }

    bla[0] = gh;

    return;
}
